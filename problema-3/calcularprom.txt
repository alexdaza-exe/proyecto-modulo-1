Algoritmo calcularprom
	cal1: Real
	cal2: Real
	cal3: Real
	cal4: Real
	cal5: Real
	promedio: Real
	definir promedio Como Real
	escribir "bienvenido al programa de promedios, para empezar, presione cualquier tecla"
	Esperar tecla
	escribir "Ingrese la calificación 1°"
	leer cal1
	escribir "Ingrese la calificación 2°"
	leer cal2
	escribir "Ingrese la calificación 3°"
	leer cal3
	escribir "Ingrese la calificación 4°"
	leer cal4
	escribir "Ingrese la calificación 5°"
	leer cal5
	promedio = (cal1+cal2+cal3+cal4+cal5)/5
	escribir "La calificación final del alumno es " promedio
	si promedio >= 70 Entonces
		escribir "El alumno aprobó, ¡enhorabuena!"
	sino
		escribir "El alumno reprobó"
	FinSi
FinAlgoritmo
